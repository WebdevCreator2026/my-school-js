//LA FONCTION POUR AJOUTER LES INFORMATIONS DES ETUDIANTS
void Ajouter(){
       struct Etudiant *temp;
       temp = (struct Etudiant *) malloc(sizeof(struct Etudiant));
       system("cls");
       printf("ENTREZ LES INFORMATIONS DE L'ETUDIANT\n");
       printf("==========================================\n");
       temp->numero_etudiant = ++numero;
       printf("ENTREZ LE NOM: ");
       scanf("%s",temp->nom);
       printf("ENTREZ LE PRENOM: ");
       scanf("%s",temp->prenom);
       printf("ENTREZ LA CLASSE: ");
       scanf("%s",temp->classe);
       printf("ENTREZ LE POURCENTAGE: ");
       scanf("%lf",&temp->pourcentage);
       temp->suivant = NULL;
       if(head==NULL)
          head = temp;
       else{
              struct Etudiant *p;
              p = head;
              while(p->suivant!=NULL)
                     p = p->suivant;
              p->suivant = temp;
       }

       system("cls");
}
//LA FONCTION POUR AFFICHER LES INFORMATIONS DES ETUDIANTS
void Afficher(){
struct Etudiant *temp;
temp = head;
system("cls");
if(temp==NULL)
       printf("LA LISTE EST VIDE\n");
else{
       while(temp!=NULL){
              printf("====================================\n");
              printf("NO : %d\n",temp->numero_etudiant);
              printf("NOM: %s\n",temp->nom);
              printf("PRENOM: %s\n",temp->prenom);
              printf("CLASSE: %s\n",temp->classe);
              printf("POURCENTAGE:%.2lf\n",temp->pourcentage);
              printf("====================================\n");
              temp = temp->suivant;
       }
}
printf("APPUYE SUR N'IMPORTE QUEL BUTTON POUR CONTINUER\n");
system("pause>0");
system("cls");

}

//LA FONCTION DE LA SUPPRESSION
void Supprimer(){
//INSTRUCTIONS POUR LA SUPPRESSION
int position,i;
struct Etudiant *p,*q;

system("cls");
printf("DE QUELLE POSITION VOUS VOULEZ SUPPRIMER? ");
scanf("%d",&position);
system("cls");
if(head==NULL)
       printf("LA SUPPRESSION IMPOSSIBLE CAR LA LISTE EST VIDE\n");
else if(position>Nombre_Etudiant())
       printf("SUPPRESSION IMPOSSIBLE CAR IL Y A %d ETUDIANT%c\n"
              ,Nombre_Etudiant(),(Nombre_Etudiant()==0 || Nombre_Etudiant()==1) ?
              ' ':'S');
else if(position==1){
  p = head;
  free(p);
  head = NULL;
}
else{

       p = head;
       i=1;
       while(i<position-1){
         p = p->suivant;
         i++;
       }
       q = p->suivant;
       p->suivant = q->suivant;
       q->suivant = NULL;
       free(q);
 }
}

//LA FONCTION DE VOIR LE NOMBRE DES ETUDIANT DANS LA LISTE
int Nombre_Etudiant(){
int nombre_etudiants=0;
struct Etudiant *temp;
temp = head;

  while(temp!=NULL){
    nombre_etudiants++;
    temp = temp->suivant;
  }
 return nombre_etudiants;

}
